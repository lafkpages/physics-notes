% Black body radiation curves for Wien's Law
% Simplified approximation of Planck's law B(x) ~ x^{-5} / (exp(a/x) - 1)
% using f(x) = C * x^{-3} * exp(-a/x) for pgfplots numerical stability.
% Same C for all curves guarantees non-crossing
\begin{tikzpicture}
  \begin{axis}[
      xmin=0, xmax=5, ymin=0, ymax=5,
      axis x line=bottom, axis y line=left,
      xlabel=$\lambda$, ylabel=$L$,
      xtick=\empty, ytick=\empty,
      extra x ticks={1.00, 1.30, 1.70},
      extra x tick labels={$\lambda_1$, $\lambda_2$, $\lambda_3$},
      extra x tick style={grid=major, grid style={dashed}},
      samples=200,
    ]

    % T1 (hottest): a=3.0, peak at 1.00, height ~4.5
    \addplot[thick, domain=0.1:5] {90 * x^(-3) * exp(-3.0/x)};
    % T2: a=3.9, peak at 1.30, height ~2.0
    \addplot[thick, domain=0.1:5] {90 * x^(-3) * exp(-3.9/x)};
    % T3 (coolest): a=5.1, peak at 1.70, height ~0.9
    \addplot[thick, domain=0.1:5] {90 * x^(-3) * exp(-5.1/x)};

    % Curve labels
    \node at (axis cs:1.0,4.8) {$T_1$};
    \node at (axis cs:1.3,2.3) {$T_2$};
    \node at (axis cs:1.7,1.2) {$T_3$};
    \node[anchor=west] at (axis cs:3.2,4.5) {$T_1 > T_2 > T_3$};
  \end{axis}
\end{tikzpicture}
